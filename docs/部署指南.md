## 数据库部署

### 安装 MySQL 8.0

```bash
# Ubuntu/Debian
sudo apt install mysql-server -y

# CentOS/RHEL
sudo yum install mysql-server -y
sudo systemctl start mysqld
sudo systemctl enable mysqld

# 验证安装
mysql --version
```

### 初始化数据库

```bash
# 登录 MySQL
sudo mysql -u root -p

# 创建数据库和用户
CREATE DATABASE `brew-now` CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
CREATE USER 'brewnow_user'@'localhost' IDENTIFIED BY 'your_secure_password';
GRANT ALL PRIVILEGES ON `brew-now`.* TO 'brewnow_user'@'localhost';
FLUSH PRIVILEGES;
EXIT;

# 导入数据库脚本
mysql -u brewnow_user -p brew-now < /path/to/project/sql/brew-now.sql
```

### 数据库优化建议

```ini
# 在 /etc/mysql/mysql.conf.d/mysqld.cnf 中添加以下配置
[mysqld]
# 连接数
max_connections = 200

# 缓存配置
innodb_buffer_pool_size = 1G
query_cache_size = 64M

# 日志配置
slow_query_log = 1
slow_query_log_file = /var/log/mysql/slow.log
long_query_time = 2

# 字符集
character-set-server = utf8mb4
collation-server = utf8mb4_unicode_ci
```

## MinIO 部署

### 安装 MinIO

```bash
# 下载 MinIO
wget https://dl.min.io/server/minio/release/linux-amd64/minio
chmod +x minio
sudo mv minio /usr/local/bin/

# 创建数据目录
sudo mkdir -p /data/minio
```

### 配置 MinIO 服务

```bash
# 创建环境变量文件
sudo tee /etc/default/minio << EOF
MINIO_ROOT_USER=minioadmin
MINIO_ROOT_PASSWORD=your_secure_minio_password
MINIO_VOLUMES="/data/minio"
MINIO_OPTS="--console-address :9001"
EOF

# 创建 systemd 服务
sudo tee /etc/systemd/system/minio.service << EOF
[Unit]
Description=MinIO
Documentation=https://docs.min.io
Wants=network-online.target
After=network-online.target

[Service]
WorkingDirectory=/usr/local/

User=minio-user
Group=minio-user
ProtectProc=invisible

EnvironmentFile=/etc/default/minio
ExecStartPre=/bin/bash -c "if [ -z \"\${MINIO_VOLUMES}\" ]; then echo \"Variable MINIO_VOLUMES not set\"; exit 1; fi"

ExecStart=/usr/local/bin/minio server \$MINIO_VOLUMES \$MINIO_OPTS

Restart=always
LimitNOFILE=65536

[Install]
WantedBy=multi-user.target
EOF

# 创建 minio 用户
sudo useradd -r minio-user -s /sbin/nologin
sudo chown -R minio-user:minio-user /data/minio

# 启动服务
sudo systemctl daemon-reload
sudo systemctl enable minio
sudo systemctl start minio
```

### 创建存储桶

访问 MinIO 控制台 `http://your-server:9001`，使用配置的账号密码登录，创建名为 `brew-now` 的存储桶，并设置访问策略为公开读取。

sudo nano /etc/redis/redis.conf

# 修改以下配置

## 后端部署

### 1. 获取项目代码

```bash
cd /opt
sudo mkdir -p brew-now
sudo chown $USER:$USER brew-now
cd brew-now

git clone https://github.com/your-username/brew-now.git .
```

### 2. 配置环境变量

```bash
# 复制环境变量文件
cp .env.example .env

# 编辑环境变量
nano .env
```

修改以下关键配置：

```env
# 数据库配置
MYSQL_ROOT_PASSWORD=your_secure_password
MYSQL_DATABASE=brew-now
MYSQL_USER=brewnow_user
MYSQL_PASSWORD=your_secure_password

# MinIO 配置
MINIO_ENDPOINT=http://127.0.0.1:9000
MINIO_PUBLIC_ENDPOINT=http://your-domain.com/minio
MINIO_ACCESS_KEY=minioadmin
MINIO_SECRET_KEY=your_secure_minio_password
MINIO_BUCKET=brew-now

# JWT 密钥（生成方式：openssl rand -base64 32）
JWT_SECRET=your-generated-jwt-secret

# Redis 配置（如启用）
REDIS_CACHE_ENABLED=true
REDIS_HOST=127.0.0.1
REDIS_PORT=6379
REDIS_PASSWORD=your_secure_redis_password
```

### 3. 修改 application.yml

确保后端配置正确指向生产环境的数据库和 MinIO：

```bash
nano backend/src/main/resources/application.yml
```

关键配置项：

```yaml
spring:
  datasource:
    url: jdbc:mysql://localhost:3306/brew-now?useUnicode=true&characterEncoding=utf8&useSSL=false&serverTimezone=Asia/Shanghai&allowPublicKeyRetrieval=true
    username: brewnow_user
    password: ${MYSQL_PASSWORD}

jwt:
  secret: ${JWT_SECRET}
  expire: 7200

minio:
  endpoint: ${MINIO_ENDPOINT}
  public-endpoint: ${MINIO_PUBLIC_ENDPOINT}
  access-key: ${MINIO_ACCESS_KEY}
  secret-key: ${MINIO_SECRET_KEY}
  bucket: ${MINIO_BUCKET}
```

### 4. 构建项目

```bash
cd backend

# 使用 Maven 构建
mvn clean package -DskipTests

# 构建产物位于
# target/brew-now-backend-1.0.0.jar
```

### 5. 启动后端

```bash
# 方式一：直接使用环境变量启动
cd /opt/brew-now
export $(cat .env | grep -v ^# | xargs)
java -jar backend/target/brew-now-backend-1.0.0.jar

# 方式二：使用 nohup 后台运行
nohup java -jar backend/target/brew-now-backend-1.0.0.jar > backend/logs/app.log 2>&1 &
```

## Nginx 配置

### 1. 复制配置文件

```bash
# 复制项目中的 Nginx 配置
sudo cp nginx/brew-now.conf /etc/nginx/sites-available/brew-now

# 创建软链接
sudo ln -s /etc/nginx/sites-available/brew-now /etc/nginx/sites-enabled/
```

### 2. 修改配置

```bash
sudo nano /etc/nginx/sites-available/brew-now
```

关键配置项：

```nginx
server {
    listen 80;
    server_name your-domain.com;  # 替换为你的域名

    # 前端静态文件目录
    root /opt/brew-now/frontend/dist;
    index index.html;

    # 前端 SPA 路由支持
    location / {
        try_files $uri $uri/ /index.html;
    }

    # 反向代理到后端
    location /api/ {
        proxy_pass http://127.0.0.1:8080;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # MinIO 代理
    location /minio/ {
        rewrite ^/minio/(.*)$ /$1 break;
        proxy_pass http://127.0.0.1:9000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
```

### 3. 测试并重启 Nginx

```bash
# 测试配置
sudo nginx -t

# 重启 Nginx
sudo systemctl restart nginx
sudo systemctl enable nginx
```

## 系统服务配置

### 创建后端 systemd 服务

```bash
sudo tee /etc/systemd/system/brewnow-backend.service << EOF
[Unit]
Description=BrewNow Backend Service
After=network.target mysql.service redis.service minio.service

[Service]
Type=simple
User=www-data
WorkingDirectory=/opt/brew-now
EnvironmentFile=/opt/brew-now/.env
ExecStart=/usr/bin/java -jar /opt/brew-now/backend/target/brew-now-backend-1.0.0.jar
Restart=on-failure
RestartSec=10
SuccessExitStatus=143

# 日志
StandardOutput=journal
StandardError=journal
SyslogIdentifier=brewnow-backend

[Install]
WantedBy=multi-user.target
EOF

# 重新加载 systemd
sudo systemctl daemon-reload

# 启动服务
sudo systemctl enable brewnow-backend
sudo systemctl start brewnow-backend

# 查看状态
sudo systemctl status brewnow-backend

# 查看日志
sudo journalctl -u brewnow-backend -f
```

### 服务管理命令

```bash
# 启动所有服务
sudo systemctl start mysql redis minio brewnow-backend nginx

# 停止所有服务
sudo systemctl stop brewnow-backend nginx

# 重启后端服务
sudo systemctl restart brewnow-backend

# 查看服务状态
sudo systemctl status mysql redis minio brewnow-backend nginx
```

## HTTPS 配置

### 使用 Let's Encrypt 免费证书

```bash
# 安装 certbot
sudo apt install certbot python3-certbot-nginx -y

# 获取证书
sudo certbot --nginx -d your-domain.com

# 自动续期
sudo certbot renew --dry-run
```

### 更新 Nginx 配置

certbot 会自动更新 Nginx 配置以支持 HTTPS。配置完成后：

```nginx
server {
    listen 443 ssl http2;
    server_name your-domain.com;

    ssl_certificate /etc/letsencrypt/live/your-domain.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/your-domain.com/privkey.pem;

    # ... 其他配置保持不变
}

# HTTP 自动跳转到 HTTPS
server {
    listen 80;
    server_name your-domain.com;
    return 301 https://\$server_name\$request_uri;
}
```

## 备份与恢复

### 数据库备份

```bash
# 创建备份目录
sudo mkdir -p /opt/brew-now/backups

# 备份数据库
mysqldump -u brewnow_user -p brew-now > /opt/brew-now/backups/brew-now-$(date +%Y%m%d).sql

# 压缩备份
mysqldump -u brewnow_user -p brew-now | gzip > /opt/brew-now/backups/brew-now-$(date +%Y%m%d).sql.gz
```

### 数据库恢复

```bash
# 恢复数据库
mysql -u brewnow_user -p brew-now < /opt/brew-now/backups/brew-now-20260101.sql

# 恢复压缩备份
gunzip < /opt/brew-now/backups/brew-now-20260101.sql.gz | mysql -u brewnow_user -p brew-now
```

### 自动备份脚本

```bash
# 创建备份脚本
cat > /opt/brew-now/backups/backup.sh << 'EOF'
#!/bin/bash
BACKUP_DIR="/opt/brew-now/backups"
DATE=$(date +%Y%m%d_%H%M%S)
DB_USER="brewnow_user"
DB_NAME="brew-now"
DB_PASS="your_secure_password"

# 备份数据库
mysqldump -u $DB_USER -p$DB_PASS $DB_NAME | gzip > $BACKUP_DIR/brew-now-$DATE.sql.gz

# 删除 30 天前的备份
find $BACKUP_DIR -name "brew-now-*.sql.gz" -mtime +30 -delete

echo "Backup completed: $DATE"
EOF

chmod +x /opt/brew-now/backups/backup.sh

# 添加 cron 定时任务（每天凌晨 2 点备份）
crontab -e
# 添加以下行：
0 2 * * * /opt/brew-now/backups/backup.sh
```

## 故障排查

### 后端无法启动

```bash
# 检查 Java 版本
java -version

# 检查端口占用
sudo lsof -i :8080

# 查看详细错误日志
sudo journalctl -u brewnow-backend -n 100 --no-pager
```

### 数据库连接失败

```bash
# 检查 MySQL 服务状态
sudo systemctl status mysql

# 测试数据库连接
mysql -u brewnow_user -p -e "SELECT 1"

# 检查数据库配置
cat backend/src/main/resources/application.yml | grep datasource
```

### 前端页面空白

```bash
# 检查 Nginx 配置
sudo nginx -t

# 检查前端文件是否存在
ls -la /opt/brew-now/frontend/dist/

# 查看 Nginx 错误日志
sudo tail -f /var/log/nginx/error.log
```

### MinIO 无法访问

```bash
# 检查 MinIO 服务状态
sudo systemctl status minio

# 检查存储桶
mc alias set myminio http://127.0.0.1:9000 minioadmin your_password
mc ls myminio/brew-now
```

### 常见错误码

| 错误 | 可能原因 | 解决方法 |
|------|----------|----------|
| 502 Bad Gateway | 后端服务未启动 | 启动后端服务 `sudo systemctl start brewnow-backend` |
| 504 Gateway Timeout | 后端响应超时 | 检查后端日志，优化慢查询 |
| 403 Forbidden | 权限问题 | 检查文件权限和 Nginx 配置 |
| 数据库连接失败 | MySQL 配置错误 | 检查 application.yml 中的数据库配置 |

## 性能优化建议

### JVM 优化

```bash
# 修改启动参数
java -Xms512m -Xmx1024m \
     -XX:+UseG1GC \
     -XX:MaxGCPauseMillis=200 \
     -jar backend/target/brew-now-backend-1.0.0.jar
```

### MySQL 优化

- 为常用查询字段添加索引
- 启用查询缓存
- 调整 innodb_buffer_pool_size

### Nginx 优化

```nginx
# 启用 gzip 压缩
gzip on;
gzip_types text/plain text/css application/json application/javascript text/xml;
gzip_min_length 1000;

# 启用缓存
location ~* \.(jpg|jpeg|png|gif|ico|css|js)$ {
    expires 30d;
    add_header Cache-Control "public, immutable";
}
```
